import gradio as gr
from apscheduler.schedulers.background import BackgroundScheduler
from huggingface_hub import HfApi, CommitOperationAdd
from datetime import datetime
import json, pathlib, subprocess, sys

REPO   = "kaidjuric/mtm-resource-sweep"
SPACE  = pathlib.Path(".")
LOG    = []

def log(msg):
    line = f"[{datetime.utcnow():%H:%M:%S}] {msg}"
    LOG.append(line)
    return "\n".join(LOG[-200:])

def run_weekly_arxiv(progress=gr.Progress()):
    progress(0.1, "fetching cs.SD · cs.LG · eess.AS · cs.MM")
    out = SPACE / "reports" / f"weekly_{datetime.today():%Y-%m-%d}_arxiv.md"
    out.parent.mkdir(exist_ok=True)
    # worker lives in mtm_sweep/arxiv_sweep.py — called as a module
    subprocess.run([sys.executable, "-m", "mtm_sweep.arxiv_sweep",
                    "--out", str(out)], check=True)
    progress(1.0, "pushing to Space")
    HfApi().create_commit(REPO, [CommitOperationAdd("reports/"+out.name,
                       pathlib.Path(out).read_bytes())],
                       commit_message=f"sweep: {out.name}")
    return log(f"weekly sweep → {out.name}")

with gr.Blocks(title="MTM Resource Sweep", theme=gr.themes.Soft()) as demo:
    gr.Markdown("# 🎼 MTM Resource Sweep\nZero-config. Weekly arXiv · monthly GitHub · quarterly foundation.")
    run  = gr.Button("Run weekly sweep now", variant="primary")
    logb = gr.Textbox(label="live cycle log", lines=20, interactive=False)
    run.click(run_weekly_arxiv, outputs=logb, every=2)

# schedule the worker — Mon 09:00 CET = 08:00 UTC (winter) / 07:00 UTC (DST)
sched = BackgroundScheduler(daemon=True)
sched.add_job(run_weekly_arxiv, "cron", day_of_week="mon", hour=7, minute=0)
sched.start()

if __name__ == "__main__":
    demo.launch(server_name="0.0.0.0", server_port=7860)

def render(name: str, **kwargs) -> str:
    tmpl = globals()[name]
    return Template(tmpl).substitute(**kwargs)

def hash_prompt(name: str) -> str:
    return hashlib.sha256(globals()[name].encode()).hexdigest()[:12]
